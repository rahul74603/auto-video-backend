name: Trending Radar (niche trend miner)

on:
  schedule:
    - cron: '40 2 * * *'   # 02:40 UTC = 8:10 AM IST
  workflow_dispatch:

jobs:
  radar:
    runs-on: ubuntu-latest

    defaults:
      run:
        working-directory: ai_backend

    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: 🛑 Check Automation Switch (trending_radar)
        id: automation_check
        env:
          SERVICE_ACCOUNT_JSON: ${{ secrets.SERVICE_ACCOUNT_JSON }}
        run: |
          npm install --ignore-scripts --legacy-peer-deps firebase-admin 2>&1 | tail -n 3 || true
          node check_automation.js trending_radar || STATUS=$?
          if [ "$STATUS" = "78" ]; then
            echo "enabled=false" >> $GITHUB_OUTPUT
            echo "⏸️ trending_radar paused — skipping workflow"
            exit 0
          else
            echo "enabled=true" >> $GITHUB_OUTPUT
          fi

      - name: Setup Node.js
        if: steps.automation_check.outputs.enabled == 'true'
        uses: actions/setup-node@v4
        with:
          node-version: '20'

      - name: Install Dependencies
        if: steps.automation_check.outputs.enabled == 'true'
        run: npm install --legacy-peer-deps

      - name: Run Trending Radar
        if: steps.automation_check.outputs.enabled == 'true'
        env:
          SERVICE_ACCOUNT_JSON: ${{ secrets.SERVICE_ACCOUNT_JSON }}
        run: node trending_radar.js
