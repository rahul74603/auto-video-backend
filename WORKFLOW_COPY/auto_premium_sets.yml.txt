# =====================================================================
#  Auto Premium Set Generator — 4 sets/day, rotated across every section
#  ACTIVE WORKFLOW REQUIRES MANUAL COPY:
#  Copy to .github/workflows/auto_premium_sets.yml
#
#  Four slots (IST): 2:30 AM, 8:30 AM, 2:30 PM, 8:30 PM.
#  Each run creates one set. Four slots cover the 16 default exam sections
#  in four days (custom system_settings/auto_premium_sets.blueprints included).
#  Required secrets: SERVICE_ACCOUNT_JSON, GEMINI_API_KEY,
#                    SETS_GEMINI_API_KEY (only premium-set fallback).
# =====================================================================
name: Auto Premium Set Generator

on:
  schedule:
    - cron: '0 21 * * *' # 2:30 AM IST — slot 0
    - cron: '0 3 * * *'  # 8:30 AM IST — slot 1
    - cron: '0 9 * * *'  # 2:30 PM IST — slot 2
    - cron: '0 15 * * *' # 8:30 PM IST — slot 3
  workflow_dispatch:
    inputs:
      slot:
        description: 'Daily slot to run now (0-3)'
        required: false
        default: '0'

permissions:
  contents: read

concurrency:
  group: auto-premium-sets
  cancel-in-progress: false

jobs:
  generate:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
          cache-dependency-path: ai_backend/package-lock.json
      - name: Install ai_backend dependencies
        working-directory: ai_backend
        run: npm ci --ignore-scripts
      - name: Generate one set for this slot
        working-directory: ai_backend
        env:
          SERVICE_ACCOUNT_JSON: ${{ secrets.SERVICE_ACCOUNT_JSON }}
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          SETS_GEMINI_API_KEY: ${{ secrets.SETS_GEMINI_API_KEY }}
          MANUAL_SLOT: ${{ github.event.inputs.slot }}
          SCHEDULE_CRON: ${{ github.event.schedule }}
        run: |
          SLOT="${MANUAL_SLOT:-0}"
          case "$SCHEDULE_CRON" in
            '0 21 * * *') SLOT=0 ;;
            '0 3 * * *')  SLOT=1 ;;
            '0 9 * * *')  SLOT=2 ;;
            '0 15 * * *') SLOT=3 ;;
          esac
          node run_auto_premium_sets.js --slot="$SLOT"
