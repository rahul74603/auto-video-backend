name: 🧩 Job Extractor Sidecar (Report-Only)

# Isko .github/workflows/job_extractor.yml naam se upload karna hai.
# Purane govt_jobs_scraper.yml se iska koi lena-dena nahi — ye alag
# chalta hai, sirf report banata hai, drafts NAHI.

on:
  schedule:
    - cron: '45 1 * * *'   # roz ~07:15 IST (main scraper ke baad)
  workflow_dispatch:        # Manual Run bhi ho sakta hai

jobs:
  extract:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - name: 📥 Checkout
        uses: actions/checkout@v4

      - name: 🐍 Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: 📦 Install Dependencies
        run: pip install -q pdfplumber extruct requests beautifulsoup4 lxml firebase-admin

      - name: 🧪 Unit Tests (golden usool — pehle test, phir run)
        working-directory: ai_backend/job_extractor
        run: python -m unittest test_extract.py

      - name: 🧩 Run Sidecar (report-only)
        working-directory: ai_backend/job_extractor
        env:
          SERVICE_ACCOUNT_JSON: ${{ secrets.SERVICE_ACCOUNT_JSON }}
        run: python run_sidecar.py
